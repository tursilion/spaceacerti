* GPL Boot Code for Super Space Acer
* Hand Assembled - no assembler syntax honored

* Start at >8000 so >6000 is free for Editor/Assembler
    org >8000
    
* standard header
    data >aa01      * intro
    data >0000      * program count+spare
    data >800C      * powerup list
    data >8060      * program list
    data >0000      * DSR list
    data >0000      * subprogram list
    
* startup list    
    data >0000
    data >8040
    
* 8010 - program entry
    data >0000      * link to next
    data >8030      * program start
    data >1553      * length plus 'S'
    text 'UPER SPACE ACER 2024'
    
* padding
    data >ffff
    data >ffff
    data >ffff
    
* 8030 - program start
07 20   all >20             * clear screen
bf 00   dst @>8300,>602a    * start address of CPU code
60 2a
0f f0   XML >F0             * run assembly from pointer at >8300
0b      EXIT                * reboot if somehow we return

        byte >ff
        data >ffff
        data >ffff
        data >ffff

* 8049

86 8F

bf 00   dst @>8300,>7318    * start address of CPU code
73 18
0f f0   XML >F0             * run assembly from pointer at >8300
0b      EXIT                * reboot if somehow we return

* 8056
bd 90
73 90
72      DST *>8372,*>8373	* Transfer address from data stack (72) to sub stack (73)
96 72   DECT @>8372			* decrement data stack pointer
00      RTN					* branches to next powerup routine        
        
* padding
        data >ffff
